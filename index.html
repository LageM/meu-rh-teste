async function buscarInforme() { const cpfFormatado = inputCpf.value; const cpf = cpfFormatado.replace(/\D/g, ""); const ano = document.getElementById('ano').value; const msg = document.getElementById('msg'); if (cpf.length !== 11) { msg.style.color = "red"; msg.innerText = "Digite um CPF válido."; return; } msg.className = "loading"; msg.innerText = "Localizando informe..."; try { const response = await fetch("https://webhook.fiqon.app/webhook/a107b458-7614-47a8-b36b-3ba893f1814c/41d2a753-2134-4ce5-8a15-4f6cc5fbea7b", { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ cpf: cpf, ano_calendario: ano }) }); if (!response.ok) { throw new Error("Informe não encontrado"); } const blob = await response.blob(); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `Informe_${cpf}_${ano}.pdf`; document.body.appendChild(a); a.click(); a.remove(); msg.style.color = "green"; msg.innerText = "Download iniciado!"; } catch (err) { msg.style.color = "red"; msg.innerText = "Informe não encontrado ou erro no servidor."; } }
